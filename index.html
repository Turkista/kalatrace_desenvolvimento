async function ktApi(path, options={}){
  const headers=Object.assign({'Content-Type':'application/json'},options.headers||{});
  const token=ktToken(); if(token) headers.Authorization='Bearer '+token;
  let res;
  try{
    res=await fetch(KT_API_BASE+path,{...options,headers});
  }catch(e){
    throw new Error('Não foi possível conectar à API do KALATRACE. Confira se a API FastAPI está ativa e se window.KALATRACE_API_BASE aponta para o endereço correto.');
  }
  const body=await res.json().catch(()=>({}));
  if(!res.ok){
    const error=new Error(body.detail||body.message||('HTTP '+res.status));
    error.status=res.status;
    throw error;
  }
  return body;
}

async function ktLogin(event){
  if(event)event.preventDefault();
  const form=document.getElementById('kt-auth-form');
  if(form&&!form.reportValidity())return;
  const emailInput=document.getElementById('kt-auth-email');
  const passwordInput=document.getElementById('kt-auth-password');
  const submit=document.getElementById('kt-auth-submit');
  const msg=document.getElementById('kt-auth-msg');
  if(submit.disabled)return;
  submit.disabled=true;
  msg.style.color='var(--text-secondary)';
  msg.textContent='Autenticando…';
  try{
    const data=await ktApi('/auth/login',{method:'POST',body:JSON.stringify({email:emailInput.value.trim(),password:passwordInput.value})});
    const token=data.access_token||data.session?.access_token;
    if(!token)throw new Error('A API não retornou um token de acesso. Verifique a configuração do endpoint de autenticação.');
    sessionStorage.setItem('kalatrace_access_token',token);
    const user=await ktApi('/auth/me');
    ktSetAuthIdentity(user);
    ktCloseAuth();
    await ktRefreshBackendStatus();
    // Retoma o passo que solicitou autenticação; se não houver um, abre o projeto padrão.
    const resume=window.KT_AUTH_CONTINUE;
    window.KT_AUTH_CONTINUE=null;
    if(typeof resume==='function')await resume();
    else await obEnterProject();
  }catch(e){
    if(e.status===400||e.status===401){
      msg.textContent='Não foi possível entrar. Confira o e-mail, a senha e se o e-mail da conta está confirmado.';
    }else if(e.status===403){
      msg.textContent='Esta conta não está autorizada a acessar o KALATRACE.';
    }else{
      msg.textContent=e.message||'Ocorreu um erro durante a autenticação.';
    }
    msg.style.color='var(--danger)';
    if(e.status===401||e.status===403){
      sessionStorage.removeItem('kalatrace_access_token');
      ktSetAuthIdentity(null);
    }
  }finally{
    submit.disabled=false;
  }
}

async function ktEnsureAuth(){
  if(ktToken()){
    try{
      const user=await ktApi('/auth/me');
      ktSetAuthIdentity(user);
      return true;
    }catch(e){
      sessionStorage.removeItem('kalatrace_access_token');
      ktSetAuthIdentity(null);
    }
  }
  ktOpenAuth('Sua sessão não está ativa. Entre com sua conta KALATRACE.');
  return false;
}
